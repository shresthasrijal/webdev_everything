1) FOR LOOP:

    repeat a block of code a set amount of times.

    # for loop basic
    for (initialization; condition; increment or decrement) {
        // code block to be executed
    }

    example:
    -------
    for (let i = 0; i < 5; i++) {
        console.log(i);
    }

    note: infinite loops.  a condition that is always true...

3)  loop3
    U can loop over properties of an object
    ---
    const fruit = {
    name: 'apple',
    color: 'red',
    price: 0.99
    };

    for (const prop in fruit) {
        console.log(fruit[prop]);
    }


    --- u can loop over properties inside nested object too

    const person = {
    name: 'John',
    age: 30,
    address: {
        street: '123 Main St',
        city: 'Anytown',
        state: 'CA'
    }
    };

    function isObject(obj) {
        return typeof obj === 'object' && !Array.isArray(obj) && obj !== null;
    }

    for (const prop in person) {
        if (isObject(person[prop])) {
            for (const nestedProp in person[prop]) {
            console.log(person[prop][nestedProp]);
            }
        } else {
            console.log(person[prop]);
        }
    }

5) .eaa.e..aa.eee.aeeae....
    break and continue statments in loops

    a break statement is used to exit the loop early. 
    a continue statement is used to skip the current iteration of the loop and skip
    directly tot he next one

    -----
    for (let i = 0; i < 10; i++) {
        if (i === 5) {
            break;
        }
        console.log(i);
    }

    in this case when i === 5 the loop stops


    -----
    for (let i = 0; i < 10; i++) {
        if (i === 5) {
            continue;
        }
        console.log(i);
    }
    
    in this case when i === 5 the loop skips to the next step when i == 6..
    without performing any action after continue for this iteration


    nestedloop:
    Apparently u can name loops in case u wawnt to skip or break them...

    outer: for (let i = 0; i < 3; i++) {
        middle: for (let j = 0; j < 3; j++) {
            for (let k = 0; k < 3; k++) {
                if (k === 1) continue middle; // skips to next j
                if (i === 1 && j === 1) break outer; // exits everything
                console.log(i, j, k);
            }
        }
    }
