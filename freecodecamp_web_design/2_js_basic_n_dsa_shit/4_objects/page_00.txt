    delete mutates original object
    - like literally remove a propery from existing object

    const person = {
        name: "Alice",
        age: 30,
        job: "Engineer"
    };

    delete person.job;

    console.log(person);     // { name: "Alice", age: 30 }
    console.log(person.job); // undefined

    Another method to remove properties is by using destructuring
    assignment with rest parameters. This approach doesnt delete the
    property but creates a new object without specified properties.

    const person = {
        name: "Bob",
        age: 25,
        job: "Designer",
        city: "New York"
    };

    // Create a new object without 'job' and 'city'
    const { job, city, ...cleanedPerson } = person;

    console.log(person.name);        // "Bob"
    console.log(cleanedPerson.name); // "Bob"

    // Modify the new object
    cleanedPerson.name = "Alice";

    console.log(cleanedPerson.name); // "Alice"
    console.log(person.name);        // "Bob" (unchanged!)

    Checking if an Object Has a Property
    There are 4 common ways to check

    - Object.hasOwn()
    Checks if the property directly belongs to the object.
    Only checks if propery exists not value

    const person = {
        name: "Alice",
        age: 30
    };

    console.log(Object.hasOwn(person, "name")); // true
    console.log(Object.hasOwn(person, "job"));  // false

   - hasOwnProperty()
   same as hasOwn but older syntax

    console.log(person.hasOwnProperty("name")); // true
    console.log(person.hasOwnProperty("job"));  // false
   
   - in operator
   Checks whether the property exists either directly on the object OR through its prototype.

   const person = {
        name: "Bob",
        age: 25
    };

    console.log("name" in person); // true
    console.log("job" in person);  // false

    - check against undefined

    const car = {
        brand: "Toyota",
        model: "Corolla"
    };

    console.log(car.brand !== undefined); // true
    console.log(car.color !== undefined); // false

    - using Object() without new
    Object(value) can convert primitive values into object wrappers:

        const num = 42;
        const numObj = Object(num);

        console.log(typeof numObj);
        // "object"


        Object(42)     // Number object
        Object("hi")   // String object
        Object(true)   // Boolean object
        
        note: null and undefined are empty object
        Object(null);       // {}
        Object(undefined);  // {}

    - Converting an unknown value
    Ucan use Object() when you need to make sure a value is an object:

    function toObject(value) {
        if (value === null || value === undefined) {
            return {};
        }

        if (typeof value === "object") {
            return value;
        }

        return Object(value);
    }

    console.log(toObject(null));       // {}
    console.log(toObject(true));       // Boolean object
    console.log(toObject([1, 2, 3]));  // [1, 2, 3]

    arrays are objects in JavaScript, typeof [1, 2, 3]; // "object"
