#2 (pop)
let fruits = ["apple", "banana", "orange"];
let lastFruit = fruits.pop();
console.log(fruits); // ["apple", "banana"]
console.log(lastFruit); // "orange"

pop() method removes the last element (orange) from the array and returns it.

#3 (unshift) 
this method adds one or more elements to the beginning of an array

let numbers = [2, 3];
let newLength = numbers.unshift(1);
console.log(numbers); // [1, 2, 3]
console.log(newLength); // 3

#4 (shift)
it removes the first element from an array and returns that element

let colors = ["red", "green", "blue"];
let firstColor = colors.shift();
console.log(colors); // ["green", "blue"]
console.log(firstColor); // "red"

Array destructuring is a feature in JavaScript that allows you to extract 
values from arrays and assign them to variables in a more concise and 
readable way. 

let fruits = ["apple", "banana", "orange"];

let [first, second, third] = fruits;

console.log(first);  // "apple"
console.log(second); // "banana"
console.log(third);  // "orange"

better than doing...

const first = fruits[0];
const second = fruits[1];
const third = fruits[2];

console.log(first); // "apple"
console.log(second); // "banana"
console.log(third); // "orange"

u can also skip elements in between ...

let colors = ["red", "green", "blue", "yellow"];
let [firstColor, , thirdColor] = colors;

console.log(firstColor); // "red"
console.log(thirdColor); // "blue"

U can also use default values..

let numbers = [1, 2];
let [a, b, c = 3] = numbers;

console.log(a); // 1
console.log(b); // 2
console.log(c); // 3

U can also have rest syntax denoted via ...rest
It allows you to capture the remaining elements of an array that haven’t been destructured into a new array

let fruits = ["apple", "banana", "orange", "mango", "kiwi"];
let [first, second, ...rest] = fruits;

console.log(first);  // "apple"
console.log(second); // "banana"
console.log(rest);   // ["orange", "mango", "kiwi"]

rest captures all remaining elements as a new array.
REST IS ALWAYS LAST in the destructuring pattern.

Reversing a string is a common programming task that can be accomplished in JavaScript using a combination of string and array methods.

Splitting the string into an array of characters.
Reversing the array.
Joining the characters back into a string.

methods: split(), reverse(), and join()

1) split()
divide a string into an array via split..
    - "" splits into individual characters
    - " " splits every space
    - "-" splits at each - character

    let str = "hello";
    let charArray = str.split("");
    console.log(charArray); // ["h", "e", "l", "l", "o"]

2) reverse()
array method that reverses the elements of an array in place.

let charArray = ["h", "e", "l", "l", "o"];
charArray.reverse();
console.log(charArray); // ["o", "l", "l", "e", "h"]

3) join()
method creates and returns a new string by concatenating all the elements 
in an array,

let reversedArray = ["o", "l", "l", "e", "h"];
let reversedString = reversedArray.join("");
console.log(reversedString); // "olleh"
